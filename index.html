# Example Python (pytrends + pandas) to fetch Google Trends and export CSV
from pytrends.request import TrendReq
import pandas as pd
pytrends = TrendReq(hl='en-US', tz=330)
kw = ["healthy meal delivery", "diet food"]
pytrends.build_payload(kw, cat=0, timeframe='2019-01-01 2025-01-01', geo='IN')
df = pytrends.interest_over_time()
df.to_csv('data/google_trends.csv')
